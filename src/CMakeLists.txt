cmake_minimum_required(VERSION 3.16)
project(Under20 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32)
    enable_language(RC)
    add_executable(Under20 WIN32 window.cpp media.cpp process.cpp app.rc)
    target_compile_definitions(Under20 PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
    target_include_directories(Under20 PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(Under20 PRIVATE comctl32 comdlg32 shell32 ole32 uxtheme advapi32 gdi32 uuid)
    if(MINGW)
        target_link_options(Under20 PRIVATE -municode -static -s)
    elseif(MSVC)
        target_compile_options(Under20 PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
        set_property(TARGET Under20 PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
else()
    find_package(Threads REQUIRED)
    add_executable(media-test media.cpp process.cpp tests/runner.cpp)
    target_include_directories(media-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(media-test PRIVATE Threads::Threads)
endif()
